### CUSTOMIZE IMAGE

ENABLE_BOOT_EXT4="false"
ENABLE_BOOT_VFAT="true"
FSTYPE="ext4"
ENABLE_BOOTCMD="false"
ENABLE_EFI="false"
GENERIC_KERNEL="false"
AML_EMMC="false"
ENABLE_PETITBOOT="false"
MMDEBSTRAP="true"
MINIMAL_ROOTFS="true"
EXTRA_FIRMWARE="true"
IMGSIZE="12288MB"
ENABLE_SHRINK="true"
IMG_COMPRESSION="xz"

user_function (){
	if [[ "$BOARD" != "z96a-v2" ]]; then return 0; fi

	# --- Z96A initramfs Plymouth assets (staged into /root by stage1) ---
	SRC=/root/z96a-chroot-files
	if [ -d "$SRC" ]; then
		install -m 0755 "$SRC/initramfs/hook" /etc/initramfs-tools/hooks/z96a-plymouth
		cp "$SRC/initramfs/modules" /etc/initramfs-tools/modules
		cp "$SRC/initramfs/initramfs.conf" /etc/initramfs-tools/initramfs.conf
		mkdir -p /usr/share/plymouth/themes
		cp -a "$SRC/theme/alkaid" /usr/share/plymouth/themes/
		rm -rf "$SRC"
	fi

	# --- Z96A boot/display: LOGO -> Plymouth animation -> desktop ---
	# The DRM stack is loaded by slim (the bootloader logo stays on screen).
	# Plymouth plays the Alkaid animation; we also repaint the last known
	# frame into the DRM framebuffer before/after it, so no black gap shows.
	if [ -f /etc/init.d/slim ] && ! grep -q z96a_paint_fb /etc/init.d/slim; then
		cat > /tmp/z96a-slim-helper <<'Z96A_HELPER'
paint_fb() {
	[ -r "$1" ] || return 0
	[ -e /dev/fb0 ] || return 0
	/usr/bin/gunzip -c "$1" > /dev/fb0 2>/dev/null || true
}

Z96A_HELPER
		cat > /tmp/z96a-slim-block <<'Z96A_BLOCK'
      # Z96A: bring the display stack up now (the bootloader logo stays on
      # screen until this point), then hand over through Plymouth so the
      # boot animation covers the remaining init.
      . /etc/default/z96a-anim 2>/dev/null || true
      ANIM_SECS=${Z96A_ANIM_SECONDS:-0}
      case "$ANIM_SECS" in
        ''|*[!0-9]*) ANIM_SECS=0 ;;
      esac
      [ -e /sys/module/rockchipdrm ] || {
        modprobe panel_edp 2>/dev/null || true
        modprobe rockchipdrm 2>/dev/null || true
      }
      CARD=/dev/dri/by-path/platform-display-subsystem-card
      for i in $(seq 1 100); do
        [ -e "$CARD" ] && break
        sleep 0.1
      done
      paint_fb /usr/share/plymouth/themes/alkaid/logo-bgra.bin.gz
      if [ -e "$CARD" ] && [ -x /usr/bin/plymouth ]; then
        if ! /usr/bin/plymouth --ping 2>/dev/null; then
          if [ -x /usr/sbin/plymouthd ]; then
            /usr/sbin/plymouthd --mode=boot --pid-file=/run/z96a-plymouth.pid 2>/dev/null || true
            sleep 0.2
            /usr/bin/plymouth --show-splash 2>/dev/null || true
          fi
        fi
        [ "$ANIM_SECS" -gt 0 ] && sleep "$ANIM_SECS"
        /usr/bin/plymouth quit 2>/dev/null || true
        # Do not start X while plymouthd still holds the DRM master: the first
        # X server then never sends SIGUSR1 to SLiM, which kills it after its
        # ~2 minute startup timeout and only the second attempt reaches the
        # login screen (seen on .84 as a long black gap after the animation).
        i=0
        while /usr/bin/plymouth --ping 2>/dev/null; do
          i=$((i + 1))
          [ "$i" -ge 50 ] && break
          sleep 0.1
        done
        paint_fb /usr/share/plymouth/themes/alkaid/last-bgra.bin.gz
        sleep 0.3
      fi
Z96A_BLOCK
		awk -v helper=/tmp/z96a-slim-helper -v block=/tmp/z96a-slim-block '
			BEGIN { while ((getline l < helper) > 0) h = h l "\n"; while ((getline l < block) > 0) b = b l "\n" }
			/^      # Z96A: load the display stack/ { next }
			/^      # stays on screen instead of being blanked by DRM early in boot\./ { next }
			/^      modprobe panel_edp 2>\/dev\/null \|\| true$/ { next }
			/^      modprobe rockchipdrm 2>\/dev\/null \|\| true$/ { next }
			/^case \$1 in$/ && !h_done { printf "%s", h; h_done=1 }
			/^      start-stop-daemon --start --quiet \$SSD_START_ARGS$/ && !b_done { printf "%s", b; b_done=1 }
			{ print }
		' /etc/init.d/slim > /tmp/z96a-slim.new
		cat /tmp/z96a-slim.new > /etc/init.d/slim
		chmod +x /etc/init.d/slim
	fi
	# Let X take its own VT (vt7).  Keeping X on plymouth's VT with
	# -novtswitch vt01 made the first X server inherit the splash VT/KMS
	# state: it finished initialisation but never signalled SLiM, so SLiM
	# killed it after ~130 s and only the retry showed the login screen.
	# Without -novtswitch the X switch flushes that state and the first
	# server comes up ~20 s after boot.
	if [ -f /etc/slim.conf ]; then
		sed -i 's|^xserver_arguments .*|xserver_arguments   -dpi 144 -nolisten tcp -deferglyphs 16|' /etc/slim.conf
		sed -i 's|^current_theme.*|current_theme       z96a-alkaid|' /etc/slim.conf
	fi
	if [ -x /usr/sbin/plymouth-set-default-theme ]; then
		plymouth-set-default-theme alkaid >/dev/null 2>&1 || true
	fi
	# Legacy init ordering: the startpar/insserv graph pulled rpcbind /
	# nfs-common into boot on this board; the verified boot path uses
	# plain filename ordering.
	rm -f /etc/init.d/.depend.boot /etc/init.d/.depend.start /etc/init.d/.depend.stop
	chmod +x /etc/init.d/z96a-drm-guard 2>/dev/null || true
	update-rc.d z96a-drm-guard defaults >/dev/null 2>&1 || true

	# 2) mtp-probe ships as an overlay file (libmtp-runtime is not installed)
	chmod +x /usr/lib/udev/mtp-probe 2>/dev/null || true

	# 3) hardware RNG is built into the kernel; the software entropy daemon is moot
	update-rc.d -f haveged disable >/dev/null 2>&1 || true

	# 4) ntpsec has no ntpd on this image (chrony provides time) and its init
	#    script shipped without the executable bit
	chmod +x /etc/init.d/ntpsec 2>/dev/null || true
	update-rc.d -f ntpsec remove >/dev/null 2>&1 || true

	# 5) locale moved out of the deprecated /etc/environment
	if grep -q '^LANG=' /etc/environment 2>/dev/null; then
		grep '^LANG=\|^LANGUAGE=' /etc/environment >> /etc/default/locale
		sed -i '/^LANG=\|^LANGUAGE=/d' /etc/environment
	fi

}

CUSTOM="$BUILDER"
